bool clockReady() { return time(nullptr) > 1700000000; }
bool validToken(const String &value) {
  if (value.length() != 64) return false;
  for (char c : value) if (!(c >= '0' && c <= '9') && !(c >= 'a' && c <= 'f')) return false;
  return true;
}
bool validPublicKey(const String &key) {
  if (key.startsWith("sb_publishable_")) return key.length() < 512;
  int first = key.indexOf('.'), second = key.indexOf('.', first + 1);
  if (first < 0 || second < 0) return false;
  String body = key.substring(first + 1, second);
  body.replace('-', '+'); body.replace('_', '/');
  while (body.length() % 4) body += '=';
  unsigned char buffer[768]; size_t length = 0;
  if (mbedtls_base64_decode(buffer, sizeof(buffer) - 1, &length, (const unsigned char *)body.c_str(), body.length()) != 0) return false;
  buffer[length] = 0;
  JsonDocument document;
  return !deserializeJson(document, buffer) && String(document["role"] | "") == "anon";
}
int clockMinutes(const String &value) {
  if (value.length() != 5 || value[2] != ':' || !isDigit(value[0]) || !isDigit(value[1]) || !isDigit(value[3]) || !isDigit(value[4])) return -1;
  int hour = value.substring(0, 2).toInt(), minute = value.substring(3).toInt();
  return hour < 24 && minute < 60 ? hour * 60 + minute : -1;
}

class ProvisionCallbacks : public BLECharacteristicCallbacks {
  String id, assembled;
  int nextPart = 0, totalParts = 0;
  void onWrite(BLECharacteristic *characteristic) override {
    String packet = characteristic->getValue();
    int one = packet.indexOf(':'), two = packet.indexOf(':', one + 1), three = packet.indexOf(':', two + 1);
    if (one != 36 || two < 0 || three < 0 || packet.length() > 220) return;
    String incomingId = packet.substring(0, one);
    int index = packet.substring(one + 1, two).toInt(), count = packet.substring(two + 1, three).toInt();
    if (count < 1 || count > 13 || index < 0 || index >= count) return;
    if (index == 0) { id = incomingId; assembled = ""; nextPart = 0; totalParts = count; }
    if (id != incomingId || index != nextPart || count != totalParts) return;
    assembled += packet.substring(three + 1); nextPart++;
    if (assembled.length() > 2048) { assembled = ""; return; }
    if (nextPart == totalParts) {
      JsonDocument check;
      if (!deserializeJson(check, assembled) && String(check["request_id"] | "") == id) {
        ProvisionMessage message; snprintf(message.json, sizeof(message.json), "%s", assembled.c_str());
        xQueueOverwrite(provisionQueue, &message);
      }
      assembled = "";
    }
  }
};

void beginBluetooth() {
  BLEDevice::init("EcoTrack-" + deviceId); BLEDevice::setMTU(247);
  BLEDevice::setSecurityCallbacks(new SecurityCallbacks());
  // Bonding + LE Secure Connections, without PIN/MITM authentication.
  BLESecurity::setAuthenticationMode(true, false, true);
  BLESecurity::setCapability(ESP_IO_CAP_NONE);
  BLEServer *server = BLEDevice::createServer(); server->setCallbacks(new ServerCallbacks());
  BLEService *service = server->createService(SERVICE_UUID);
  infoCharacteristic = service->createCharacteristic(INFO_UUID, BLECharacteristic::PROPERTY_READ);
  infoCharacteristic->setCallbacks(new InfoCallbacks());
  #if defined(CONFIG_BLUEDROID_ENABLED)
  BLECharacteristic *writer = service->createCharacteristic(WIFI_UUID, BLECharacteristic::PROPERTY_WRITE);
  writer->setAccessPermissions(ESP_GATT_PERM_WRITE_ENC);
#else
  BLECharacteristic *writer = service->createCharacteristic(WIFI_UUID, BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_WRITE_ENC);
#endif
  writer->setCallbacks(new ProvisionCallbacks());
  #if defined(CONFIG_BLUEDROID_ENABLED)
  BLECharacteristic *capture = service->createCharacteristic(CAPTURE_UUID, BLECharacteristic::PROPERTY_READ);
  capture->setAccessPermissions(ESP_GATT_PERM_READ_ENC);
  #else
  BLECharacteristic *capture = service->createCharacteristic(CAPTURE_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_READ_ENC);
  #endif
  capture->setCallbacks(new CaptureCallbacks());
  service->start();
  BLEAdvertising *advertising = BLEDevice::getAdvertising();
  advertising->addServiceUUID(SERVICE_UUID); advertising->setScanResponse(true); advertising->start();
}
void applyConfig(JsonVariantConst config) {
  if (config["interval_seconds"].is<unsigned long>()) {
    unsigned long seconds = config["interval_seconds"];
    if (seconds >= 1 && seconds <= 86400) {
      intervalMs = seconds * 1000UL;
      if (prefs.getULong("interval", 5000) != intervalMs) prefs.putULong("interval", intervalMs);
    }
  }
  String wake = config["wake_time"] | "", sleep = config["sleep_time"] | "";
  if ((wake.isEmpty() && sleep.isEmpty()) || (clockMinutes(wake) >= 0 && clockMinutes(sleep) >= 0)) {
    wakeTime = wake; sleepTime = sleep;
    if (prefs.getString("wake") != wake) prefs.putString("wake", wake);
    if (prefs.getString("sleep") != sleep) prefs.putString("sleep", sleep);
  }
}

decode_type_t transmitterProtocol(String protocol) {
  protocol.trim();
  decode_type_t exact = strToDecodeType(protocol.c_str());
  if (exact != UNKNOWN && IRac::isProtocolSupported(exact)) return exact;
  protocol.toLowerCase();
  if (protocol == "midea_ir" || protocol == "midea") return MIDEA;
  if (protocol == "carrier") return CARRIER_AC64;
  if (protocol == "panasonic") return PANASONIC_AC;
  if (protocol == "mitsubishi") return MITSUBISHI_AC;
  if (protocol == "samsung") return SAMSUNG_AC;
  if (protocol == "haier") return HAIER_AC;
  if (protocol == "hitachi") return HITACHI_AC;
  if (protocol == "fujitsu") return FUJITSU_AC;
  if (protocol == "toshiba") return TOSHIBA_AC;
  if (protocol == "sharp") return SHARP_AC;
  if (protocol == "tcl") return TCL112AC;
  if (protocol == "whirlpool") return WHIRLPOOL_AC;
  // Do not guess an encoder for generic_ac, koppel_ir or other unknown families.
  return exact;
}
bool switchValue(JsonVariantConst value, bool &result) {
  if (value.is<bool>()) { result = value.as<bool>(); return true; }
  String text = value.is<const char *>() ? value.as<String>() : "";
  if (text == "on") { result = true; return true; }
  if (text == "off") { result = false; return true; }
  return false;
}
bool prepareAcState(JsonVariantConst payload, stdAc::state_t &desired, String &error) {
  if (!payload.is<JsonObjectConst>() || !payload["state"].is<JsonObjectConst>()) { error = "Full AC state is required."; return false; }
  decode_type_t type = transmitterProtocol(payload["protocol"] | "");
  if (type == UNKNOWN || !IRac::isProtocolSupported(type)) {
    error = "Unsupported AC protocol. Detect/select the exact protocol in Edit or IR Profiles first."; return false;
  }
  JsonVariantConst state = payload["state"];
  bool power;
  String mode = state["mode"] | "", fan = state["fan"] | "";
  if (!switchValue(state["power"], power) ||
      (mode != "cool" && mode != "dry" && mode != "fan" && mode != "auto" && mode != "heat") ||
      (fan != "auto" && fan != "low" && fan != "medium" && fan != "high" && fan != "quiet" && fan != "silent" && fan != "turbo") ||
      state["temp"].is<bool>() || (!state["temp"].is<double>() && !state["temp"].is<long>())) {
    error = "Invalid AC power, mode, temperature or fan setting."; return false;
  }
  float temperature = state["temp"].as<float>();
  if (!isfinite(temperature) || temperature < 16 || temperature > 32) { error = "AC temperature must be 16 to 32 degrees C."; return false; }
  IRac::initState(&desired);
  desired.protocol = type;
  desired.model = haveLastAcState && lastAcState.protocol == type ? lastAcState.model : IR_AC_MODEL;
  if (!payload["model"].isNull()) {
    if (!payload["model"].is<int>() || payload["model"].as<int>() < -1 || payload["model"].as<int>() > 32767) {
      error = "Invalid AC protocol model."; return false;
    }
    desired.model = payload["model"].as<int>();
  }
  desired.power = power; desired.celsius = true; desired.degrees = temperature;
  desired.mode = IRac::strToOpmode(mode.c_str());
  desired.fanspeed = fan == "quiet" || fan == "silent" ? stdAc::fanspeed_t::kMin :
    fan == "turbo" ? stdAc::fanspeed_t::kMax : IRac::strToFanspeed(fan.c_str());
  desired.quiet = fan == "quiet" || fan == "silent";
  bool swing = false, turbo = fan == "turbo", eco = false, sleep = false;
  for (const char *field : {"swing", "turbo", "eco", "sleep"}) {
    bool value;
    if (state[field].isNull()) continue;
    if (!switchValue(state[field], value)) { error = String("Invalid AC setting: ") + field; return false; }
    if (!strcmp(field,"swing")) swing = value;
    else if (!strcmp(field,"turbo")) turbo = value;
    else if (!strcmp(field,"eco")) eco = value;
    else sleep = value;
  }
  desired.swingv = swing ? stdAc::swingv_t::kAuto : stdAc::swingv_t::kOff;
  desired.turbo = turbo; desired.econo = eco; desired.sleep = sleep ? 0 : -1;
  return true;
}
void rememberAcState(const stdAc::state_t &state) {
  lastAcState = state; haveLastAcState = true;
  JsonDocument saved;
  saved["protocol"] = typeToString(state.protocol); saved["model"] = state.model;
  saved["state"]["power"] = state.power ? "on" : "off";
  saved["state"]["temp"] = state.celsius ? state.degrees : (state.degrees - 32.0f) * 5.0f / 9.0f;
  saved["state"]["mode"] = commonMode(state.mode).length() ? commonMode(state.mode) : "auto";
  saved["state"]["fan"] = commonFan(state.fanspeed).length() ? commonFan(state.fanspeed) : "auto";
  saved["state"]["swing"] = state.swingv != stdAc::swingv_t::kOff ? "on" : "off";
  saved["state"]["turbo"] = state.turbo ? "on" : "off";
  saved["state"]["eco"] = state.econo ? "on" : "off";
  saved["state"]["sleep"] = state.sleep >= 0 ? "on" : "off";
  String json; serializeJson(saved,json);
  if (prefs.getString("ir-state") != json) prefs.putString("ir-state",json);
}
void transmitAcIntent(const String &id, JsonVariantConst payload) {
  if (!IR_TX_ENABLED) { setAck(id,false,JsonVariantConst(),"IR transmitter is disabled in this sketch."); return; }
  if (String(payload["aircon_id"] | "").isEmpty()) { setAck(id,false,JsonVariantConst(),"AC profile ID is required."); return; }
  stdAc::state_t desired; String error;
  if (!prepareAcState(payload,desired,error)) { setAck(id,false,JsonVariantConst(),error); return; }
  // Avoid recording our own outgoing signal as a physical-remote observation.
  receiver.disableIRIn();
  const stdAc::state_t *previous = haveLastAcState && lastAcState.protocol == desired.protocol ? &lastAcState : nullptr;
  bool sent = transmitter.sendAc(desired,previous);
  digitalWrite(IR_TX_PIN, IR_TX_INVERTED ? HIGH : LOW);
  delay(100); receiver.enableIRIn();
  if (!sent) { setAck(id,false,JsonVariantConst(),"This IR protocol could not be transmitted."); return; }
  rememberAcState(desired);
  JsonDocument result;
  result["delivery_status"] = "transmitted";
  result["aircon_id"] = payload["aircon_id"];
  result["protocol"] = typeToString(desired.protocol);
  result["state"].set(payload["state"]);
  // This receipt confirms an IR send attempt, not physical AC operation.
  setAck(id,true,result.as<JsonVariantConst>());
  Serial.printf("IR transmitted: %s | power: %s | %.1f C | mode: %s | fan: %s | GPIO: %u\n",
    typeToString(desired.protocol).c_str(),desired.power ? "on" : "off",desired.degrees,
    commonMode(desired.mode).c_str(),commonFan(desired.fanspeed).c_str(),IR_TX_PIN);
}

void publishCaptureMetadata(JsonVariantConst frame) {
  JsonDocument metadata;
  metadata["sequence"] = frame["sequence"];
  metadata["protocol"] = frame["protocol"];
  metadata["raw_type"] = frame["raw_type"];
  metadata["hex"] = frame["hex"];
  metadata["raw_count"] = frame["raw_count"];

  portEXIT_CRITICAL(&infoMutex);
}

void collectIR() {
  if (!receiver.decode(&decoded)) return;
  if (!decoded.overflow && !decoded.repeat && decoded.rawlen >= 4) {
    JsonDocument frame;
    frame["success"] = true; frame["protocol"] = typeToString(decoded.decode_type);
    frame["raw_type"] = typeToString(decoded.decode_type);
    frame["hex"] = "0x" + uint64ToString(decoded.value, 16);
    frame["sequence"] = ++captureSequence; frame["captured_at_ms"] = millis(); frame["boot_count"] = bootCount;
    if (clockReady()) { timeval now; gettimeofday(&now, nullptr); frame["captured_at_epoch_ms"] = (uint64_t)now.tv_sec * 1000ULL + now.tv_usec / 1000; }
    String raw;
    for (uint16_t i = 1; i < decoded.rawlen; i++) {
      if (i > 1) raw += ',';
      raw += String((uint32_t)decoded.rawbuf[i] * kRawTick);
    }
    frame["raw_data"] = raw;
    frame["raw_count"] = decoded.rawlen - 1;
    stdAc::state_t acState;
    IRac::initState(&acState);
    if (IRAcUtils::decodeToState(&decoded, &acState, haveLastAcState ? &lastAcState : nullptr)) {
      rememberAcState(acState);
      frame["state"]["power"] = acState.power;
      frame["state"]["temp"] = acState.celsius ? acState.degrees : (acState.degrees - 32.0f) * 5.0f / 9.0f;
      frame["state"]["mode"] = commonMode(acState.mode);
      frame["state"]["fan"] = commonFan(acState.fanspeed);
      frame["state"]["source"] = "ir_receiver";
      // A trailing undecoded frame must not erase a decoded OFF before the next cloud poll.
      JsonDocument lastDecoded; lastDecoded.set(frame.as<JsonVariantConst>());
      lastDecoded.remove("raw_data");
      latestDecodedCapture = ""; serializeJson(lastDecoded, latestDecodedCapture);
      Serial.printf("Decoded remote state: power %s | %.1f C | mode %s | fan %s\n",
        acState.power ? "on" : "off", frame["state"]["temp"].as<float>(),
        commonMode(acState.mode).c_str(), commonFan(acState.fanspeed).c_str());
    } else if (latestDecodedCapture.length()) {
      JsonDocument lastDecoded;
      if (!deserializeJson(lastDecoded, latestDecodedCapture))
        frame["last_decoded"].set(lastDecoded.as<JsonVariantConst>());
    }
    publishCaptureMetadata(frame.as<JsonVariantConst>());
    latestCapture = ""; serializeJson(frame, latestCapture); captureDirty = true;
    Serial.printf("Captured IR: %s, %u timings\n", typeToString(decoded.decode_type).c_str(), decoded.rawlen - 1);
    if (learningId.length()) { setAck(learningId, true, frame.as<JsonVariantConst>()); learningId = ""; }
  }
  receiver.resume();
}
void handleCommand(JsonVariantConst command) {
  String id = command["id"] | "", kind = command["kind"] | "";
  if (id.isEmpty() || id == completedCommand || id == learningId) return;
  if (kind == "learn_ir") {
    // Use a frame received after this request was created, even if it arrived during cloud delivery.
    JsonDocument recent;
    uint64_t requestedAt = command["created_at_epoch_ms"] | 0ULL;
    if (requestedAt && !deserializeJson(recent, latestCapture) && (recent["captured_at_epoch_ms"] | 0ULL) >= requestedAt) {
      setAck(id, true, recent.as<JsonVariantConst>());
    } else {
      learningId = id; learningDeadline = millis() + 6000;
      Serial.println("Learning: press the physical remote toward the IR receiver now.");
    }
  }
  else if (kind == "ping") {
    JsonDocument result; result["rssi"] = WiFi.RSSI(); result["ip_address"] = WiFi.localIP().toString();
    result["firmware_version"] = VERSION; setAck(id, true, result.as<JsonVariantConst>());
  } else if (kind == "configure") {
    applyConfig(command["payload"]);
    JsonDocument result; result["interval_ms"] = intervalMs; result["wake_time"] = wakeTime; result["sleep_time"] = sleepTime;
    setAck(id, true, result.as<JsonVariantConst>());
  } else if (kind == "ac_intent") {
    transmitAcIntent(id,command["payload"]);
  } else { setAck(id, false, JsonVariantConst(), "Unsupported cloud command."); }
}
String isoUtc(time_t seconds) {
  tm value; gmtime_r(&seconds, &value); char result[25]; strftime(result, sizeof(result), "%Y-%m-%dT%H:%M:%SZ", &value); return String(result);
}
bool exchange(const String &sleepingUntil = "") {
  if (WiFi.status() != WL_CONNECTED || !clockReady() || projectUrl.isEmpty() || !validToken(deviceToken)) return false;
  JsonDocument request;
  request["p_device_id"] = deviceId; request["p_token"] = deviceToken;
  JsonObject status = request["p_status"].to<JsonObject>();
  bool reportDue = millis() - lastReportMs >= intervalMs;
  if (reportDue) status["report_id"] = deviceId + "-" + String(bootCount) + "-" + String(reportSequence);
  status["firmware_version"] = VERSION; status["wifi_ssid"] = WiFi.SSID();
  status["ip_address"] = WiFi.localIP().toString(); status["rssi"] = WiFi.RSSI();
  status["capabilities"]["ir_receiver"] = true; status["capabilities"]["ir_transmitter"] = IR_TX_ENABLED;
  status["applied_config"]["interval_seconds"] = intervalMs / 1000;
  status["applied_config"]["wake_time"] = wakeTime; status["applied_config"]["sleep_time"] = sleepTime;
  if (sleepingUntil.length()) status["sleeping_until"] = sleepingUntil;
  JsonDocument capture, ack;
  if (captureDirty && !deserializeJson(capture, latestCapture)) status["latest_ir"].set(capture.as<JsonVariantConst>());
  if (pendingAck.length() && !deserializeJson(ack, pendingAck)) request["p_ack"].set(ack.as<JsonVariantConst>());
  String body; serializeJson(request, body);
  WiFiClientSecure tls; tls.useBuiltinCACertBundle();
  HTTPClient http;
  if (!http.begin(tls, projectUrl + "/rest/v1/rpc/ecotrack_device_exchange")) return false;
  http.setConnectTimeout(5000); http.setTimeout(5000);
  http.addHeader("Content-Type", "application/json"); http.addHeader("apikey", publicKey);
  if (publicKey.startsWith("eyJ")) http.addHeader("Authorization", "Bearer " + publicKey);
  int code = http.POST(body);
  JsonDocument response;
  bool parsed = !deserializeJson(response, http.getString()); http.end();
  if (code != 200 || !parsed || response["success"] != true) {
    cloudStatus = "error";
    lastError = code == 401 || code == 403 ? "Cloud authentication failed. Reprovision this device." : String("Supabase request failed (HTTP ") + code + ").";
    Serial.println(lastError); return false;
  }
  cloudStatus = "verified"; lastError = ""; registryId = response["node_id"] | "";
  captureDirty = false;
  if (reportDue) { lastReportMs = millis(); reportSequence++; }
  if (pendingAck.length()) { pendingAck = ""; prefs.remove("ack"); }
  applyConfig(response["config"]);
  if (saveCloudAfterVerify) {
    prefs.putString("url", projectUrl); prefs.putString("public", publicKey); prefs.putString("token", deviceToken);
    prefs.putString("node", registryId); saveCloudAfterVerify = false;
  }
  if (sleepingUntil.isEmpty() && !response["command"].isNull()) handleCommand(response["command"]);
  return true;
}
void processProvisioning() {
  ProvisionMessage message;
  if (!xQueueReceive(provisionQueue, &message, 0)) return;
  JsonDocument document;
  if (deserializeJson(document, message.json)) return;
  requestId = document["request_id"] | ""; lastError = "";
  String newSSID = document["ssid"] | "", newPassword = document["pass"] | "";
  if (String(document["device_id"] | "") != deviceId || newSSID.isEmpty() || newSSID.length() > 32 || newPassword.length() < 8 || newPassword.length() > 63) {
    wifiStatus = "error"; lastError = "Invalid Wi-Fi credentials or device identity."; return;
  }
  if (!document["supabase_url"].isNull()) {
    String url = document["supabase_url"] | "", key = document["supabase_key"] | "", token = document["device_token"] | "";
    if (!url.startsWith("https://") || url.length() > 160 || !validPublicKey(key) || !validToken(token)) {
      cloudStatus = "error"; lastError = "Invalid cloud provisioning settings."; return;
    }
    while (url.endsWith("/")) url.remove(url.length() - 1);
    projectUrl = url; publicKey = key; deviceToken = token;
    cloudStatus = "connecting"; saveCloudAfterVerify = true;
    pendingAck = ""; completedCommand = ""; prefs.remove("ack"); prefs.remove("completed");
  }
  ssid = newSSID; password = newPassword;
  if (WiFi.status() == WL_CONNECTED && WiFi.SSID() == ssid) { wifiStatus = "connected"; }
  else {
    WiFi.disconnect(); WiFi.mode(WIFI_STA); WiFi.begin(ssid.c_str(), password.c_str());
    wifiStatus = "connecting"; wifiStarted = millis();
  }
}
void manageWifi() {
  if (WiFi.status() == WL_CONNECTED) {
    if (wifiStatus != "connected") {
      wifiStatus = "connected"; lastError = "";
      configTime(UTC_OFFSET_SECONDS, 0, "pool.ntp.org", "time.google.com");
      Serial.print("ESP32 Wi-Fi IP: "); Serial.println(WiFi.localIP());
    }
    if (prefs.getString("ssid") != ssid) prefs.putString("ssid", ssid);
    if (prefs.getString("pass") != password) prefs.putString("pass", password);
    return;
  }
  if (ssid.isEmpty()) return;
  if (wifiStatus == "connecting" && millis() - wifiStarted >= 30000) { wifiStatus = "error"; lastError = "Wi-Fi connection failed. Check the 2.4 GHz network/password."; }
  if (wifiStatus == "connected") wifiStatus = "connecting";
  if (millis() - lastReconnect >= 30000) {
    lastReconnect = millis(); WiFi.begin(ssid.c_str(), password.c_str()); wifiStarted = millis();
  }
}
uint32_t secondsUntilWake() {
  int wake = clockMinutes(wakeTime), sleep = clockMinutes(sleepTime);
  if (!clockReady() || wake < 0 || sleep < 0 || wake == sleep) return 0;
  time_t now = time(nullptr); tm local; localtime_r(&now, &local);
  int seconds = local.tm_hour * 3600 + local.tm_min * 60 + local.tm_sec;
  bool awake = wake < sleep ? seconds >= wake * 60 && seconds < sleep * 60 : seconds >= wake * 60 || seconds < sleep * 60;
  if (awake) return 0;
  return (wake * 60 - seconds + 86400) % 86400;
}
void scheduledSleep() {
  if (millis() < 15000 || bluetoothConnected || learningId.length() || pendingAck.length() || saveCloudAfterVerify) return;
  uint32_t until = secondsUntilWake(); if (!until || millis() - lastSleepCheckMs < CLOUD_POLL_MS) return;
  lastSleepCheckMs = millis();
  uint32_t duration = min(until, RECHECK_SLEEP_SECONDS);
  if (WiFi.status() == WL_CONNECTED && !exchange(isoUtc(time(nullptr) + duration))) return;
  if (!secondsUntilWake() || bluetoothConnected || pendingAck.length()) return;
  Serial.printf("Sleeping %lu seconds; reconnecting to check the cloud schedule on wake.\n", (unsigned long)duration);
  Serial.flush(); receiver.disableIRIn();
  if (IR_TX_ENABLED) {
    digitalWrite(IR_TX_PIN,IR_TX_INVERTED ? HIGH : LOW);
    gpio_hold_en((gpio_num_t)IR_TX_PIN); gpio_deep_sleep_hold_en();
  }
  BLEDevice::deinit(true); WiFi.disconnect(true);
  esp_sleep_enable_timer_wakeup((uint64_t)duration * 1000000ULL); esp_deep_sleep_start();
}
void setup() {
  Serial.begin(115200); prefs.begin("ecotrack-cloud", false);
  if (IR_TX_ENABLED) {
    gpio_deep_sleep_hold_dis(); gpio_hold_dis((gpio_num_t)IR_TX_PIN);
    pinMode(IR_TX_PIN,OUTPUT); digitalWrite(IR_TX_PIN,IR_TX_INVERTED ? HIGH : LOW);
  }
  JsonDocument previous; String stateError;
  if (!deserializeJson(previous,prefs.getString("ir-state")) && prepareAcState(previous.as<JsonVariantConst>(),lastAcState,stateError)) haveLastAcState = true;
  bootCount = prefs.getUInt("boot", 0) + 1; prefs.putUInt("boot", bootCount);
  uint64_t mac = ESP.getEfuseMac(); char hardware[32];
  snprintf(hardware, sizeof(hardware), "esp32s3-%012llx", (unsigned long long)(mac & 0xFFFFFFFFFFFFULL)); deviceId = hardware;
  ssid = prefs.getString("ssid"); password = prefs.getString("pass");
  projectUrl = prefs.getString("url"); publicKey = prefs.getString("public"); deviceToken = prefs.getString("token"); registryId = prefs.getString("node");
  wakeTime = prefs.getString("wake"); sleepTime = prefs.getString("sleep"); intervalMs = prefs.getULong("interval", 5000);
  pendingAck = prefs.getString("ack"); completedCommand = prefs.getString("completed");
  provisionQueue = xQueueCreate(1, sizeof(ProvisionMessage));
  if (!provisionQueue) { Serial.println("Provisioning queue allocation failed."); while (true) delay(1000); }
  Serial.println("EcoTrack cloud IR receiver + transmitter: " + deviceId);
  receiver.enableIRIn(); beginBluetooth();
  WiFi.mode(WIFI_STA); configTime(UTC_OFFSET_SECONDS, 0, "pool.ntp.org", "time.google.com");
  if (ssid.length()) { WiFi.begin(ssid.c_str(), password.c_str()); wifiStatus = "connecting"; wifiStarted = millis(); }
  if (projectUrl.length()) cloudStatus = "connecting";
  updateBleInfo();
}
void loop() {
  processProvisioning(); manageWifi(); collectIR();
  if (advertiseAgain) { advertiseAgain = false; BLEDevice::startAdvertising(); }
  if (learningId.length() && (int32_t)(millis() - learningDeadline) >= 0) { setAck(learningId, false, JsonVariantConst(), "No new IR signal received within six seconds."); learningId = ""; }
  if (millis() - lastCloudPoll >= min(CLOUD_POLL_MS, intervalMs)) { lastCloudPoll = millis(); exchange(); }
  if (millis() - lastStatusPrint >= intervalMs) {
    lastStatusPrint = millis(); Serial.printf("Wi-Fi: %s | Supabase: %s | IR RX GPIO: %u | IR TX GPIO: %u (%s)\n", wifiStatus.c_str(), cloudStatus.c_str(), IR_RX_PIN, IR_TX_PIN, IR_TX_ENABLED ? "enabled" : "disabled");
  }
  updateBleInfo(); scheduledSleep(); delay(5);
}
